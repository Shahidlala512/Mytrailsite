async function fetchScrapedData() { const q = document.getElementById('query').value; const status = document.getElementById('status'); const grid = document.getElementById('movieGrid'); status.innerText = 'Scraping in progress...'; grid.innerHTML = ''; try { const res = await fetch(`/api/scrape?query=${encodeURIComponent(q)}`); const json = await res.json(); if (!json.success) { status.innerText = `Error: ${json.error}`; return; } if (json.data.length === 0) { status.innerText = 'No items found on target site!'; return; } status.innerText = `Found ${json.count} movies:`; json.data.forEach(item => { grid.innerHTML += `
${item.title}
📥 Get File Links
`; }); } catch (err) { status.innerText = `API Route Not Found (Check GitHub 'api' folder structure)`; } }
